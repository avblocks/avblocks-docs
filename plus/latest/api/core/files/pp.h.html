311/**
312Enumerates available hardware codecs.
313*/
314class Hardware: public primo::Reference
315{
316public:
317
318    /**
319    Scans the system for available hardware codecs and populates the device list.
320
321    @param vendor [in] Vendor filter. Only devices that match the specified vendor are queried.
322    By default <i>vendor</i> is HwVendor::None which queries all known devices on the system.
323    */
324    virtual void refresh(HwVendor::Enum vendor = HwVendor::None) = 0;
325
326    
327    /**
328    Returns an immutable list of devices that support 
329    hardware-accelerated encoding, decoding or processing.
330
331    @return A list of Device objects. 
332    The device list canont be <b>NULL</b>.
333    It is initially empty when the Hardware object is created.
334    It is populated after a call to the refresh() method.
335
336    The returned list exists only in the context of Hardware.
337    */
338    virtual HwDeviceList* devices() = 0;
339
340protected:
341
342    /**
343    This object can be destroyed only by the library. The caller must use Reference::release.
344
345    @see \ref object_management "Object Management"
346    */
347	virtual ~Hardware() {}
