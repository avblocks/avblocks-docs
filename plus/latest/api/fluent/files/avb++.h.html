 160/**

 163 * The strings are copied out of the source @c primo::error::ErrorInfo, so a @c TErrorInfo stays valid after
 164 * the source object is gone. A default-constructed @c TErrorInfo is empty.

 166class TErrorInfo {
 167    int32_t facility_{};
 168    int32_t code_{};
 169    std::string message_;
 170    std::string hint_;

2859    /// Opens, runs and closes the transcoder (file to file mode): the whole @c open().run().close() sequence in
2860    /// one call. The transcoder is closed even if @c run() throws; @c TAVBlocksException is thrown if it cannot be
2861    /// opened or fails to run.
2862    /// @code
2863    /// TTranscoder().allowDemoMode(true).addInput(in).addOutput(out).transcode();
2864    /// @endcode
2865    void transcode() {
2866        open();

2868            run();

2870            close();
2871            throw;
2872        }
